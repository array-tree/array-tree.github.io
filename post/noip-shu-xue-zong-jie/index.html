#include <bits/stdc++.h>
using namespace std;
const double eps = 1e-8;
int n;
double f[100][100];
bool tag_[1000];
bool noans = 0, ansm = 0;
void calc()//高斯消元
{
    for (int i = 1; i <= n; i++)//枚举每一列
    {
        int max_ = i;
        for (int j = 1; j <= n; j++)//选择主元这里这里么写其实是为了正确判断是无解还是有解
        {
            if (fabs(f[j][j]) > eps && j < i)//如果是在已经处理过的行中并且确定对应有值
            {
                continue;
            }
            if (fabs(f[j][i]) > fabs(f[max_][i]))//选择最大的主元
            {
                max_ = j;
            }
        }
        for (int j = 1; j <= n + 1; j++)//换到上面去
        {
            swap(f[i][j], f[max_][j]);
        }
        if (fabs(f[i][i]) < eps)//如果直接无解了
        {
            continue;
        }
        for (int k = 1; k <= n; k++)//消掉~
        {
            if (k == i)
            {
                continue;
            }
            double t = f[k][i] / f[i][i];
            for (int j = 1; j <= n + 1; j++)
            {
                f[k][j] -= t * f[i][j];
            }
        }
    }
}
int main()
{
    cin >> n;
    for (int ww = 1; ww <= n; ww++)
    {
        for (int ee = 1; ee <= n + 1; ee++)
        {
            cin >> f[ww][ee];
        }
    }
    for (int ww = 1; ww <= n; ww++)
    {
        bool tag_ = 1;
        for (int ee = 1; ee <= n; ee++)
        {
            if (fabs(f[ww][ee]) > 0)
            {
                tag_ = 0;
            }
        }
        if (tag_ && fabs(f[ww][n + 1]) < eps)
        {
            ansm = 1;
        }
        else if (tag_ && fabs(f[ww][n + 1]) > eps)
        {
            noans = 1;
        }
    }
    if (noans)
    {
        cout << -1 << "\n";
        return 0;
    }
    calc();
    for (int i = 1; i <= n; i++)
    {
        if (fabs(f[i][i]) < eps)
        {
            if (fabs(f[i][n + 1]) < eps)
            {
                ansm = 1;
            }
            else
            {
                noans = 1;
            }
        }
    }
    for (int i = 1; i <= n; i++)//这个不能写在判无解前面，否则会发生除以0之类的问题从而导致无法正确判断
    {
        f[i][n + 1] /= f[i][i];
    }
    if (noans)
    {
        cout << "-1";
        return 0;
    }
    else if (ansm)
    {
        cout << "0";
        return 0;
    }
    else
    {
        for (int ww = 1; ww <= n; ww++)
        {
            printf("x%d=%.2lf\n", ww, f[ww][n + 1]);
        }
    }
    return 0;
}

vector<int> pri;
bool not_prime[N];

void pre(int n) {
  for (int i = 2; i <= n; ++i) {
    if (!not_prime[i]) {
      pri.push_back(i);
    }
    for (int pri_j : pri) {
      if (i * pri_j > n) break;
      not_prime[i * pri_j] = true;
      if (i % pri_j == 0) {
        // i % pri_j == 0
        // 换言之，i 之前被 pri_j 筛过了
        // 由于 pri 里面质数是从小到大的，所以 i 乘上其他的质数的结果一定会被
        // pri_j 的倍数筛掉，就不需要在这里先筛一次，所以这里直接 break
        // 掉就好了
        break;
      }
    }
  }
